import re

def main():
    print("=== Password Strength Checker ===")
    password = input("Please enter a password to check: ")
    
    strength, feedback = check_password_strength(password)
    
    print(f"\nResult: {strength}")
    if feedback:
        print("Tips for improvement:")
        for tip in feedback:
            print(f"- {tip}")

def check_password_strength(password):
    score = 0
    feedback = []
    
    # 1. Check Length (8 characters or more)
    if len(password) >= 8:
        score += 1
    else:
        feedback.append("Password is too short (must be at least 8 characters).")
        
    # 2. Check for Digits
    if re.search(r"[0-9]", password):
        score += 1
    else:
        feedback.append("Add at least one number (0-9).")
        
    # 3. Check for Lowercase Letters
    if re.search(r"[a-z]", password):
        score += 1
    else:
        feedback.append("Add lowercase letters (a-z).")
        
    # 4. Check for Uppercase Letters
    if re.search(r"[A-Z]", password):
        score += 1
    else:
        feedback.append("Add uppercase letters (A-Z).")
        
    # 5. Check for Special Characters
    if re.search(r"[!@#$%^&*(),.?\":{}|<>]", password):
        score += 1
    else:
        feedback.append("Add special characters (e.g., @, #, $, !).")
    
    # Determine Final Rating
    if score == 5:
        return "Strong Password ✅", []
    elif score >= 3:
        return "Medium Strength ⚠️", feedback
    else:
        return "Weak Password ❌", feedback

if __name__ == "__main__":
    main()
